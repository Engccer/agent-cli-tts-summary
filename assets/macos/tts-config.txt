# TTS 음성 요약 설정
# 설치 위치: <에이전트 홈>/TTS-Summary/tts-config.txt (.claude / .codex / .gemini)
# "키=값" 한 줄씩. # 으로 시작하는 줄은 주석.
# 값을 바꾸면 다음 응답부터 바로 적용된다(세션 재시작 불필요).

# 사용 여부: on / off
# off면 요약 작성도, 음성 재생도 하지 않는다(요약 누락 가드도 걸리지 않는다).
enabled=on

# 말하기 속도: 1(가장 느림) ~ 10(가장 빠름). 소수점 허용(예: 7.5). 5가 보통 속도
# 5 위로는 2.5칸마다 두 배가 된다: 5=200wpm, 6=264, 7=348, 7.5=400, 8=460, 9=606, 10=800.
# 10(800wpm)은 macOS say의 포화 속도라 그 위로 올려도 더 빨라지지 않는다.
# API provider는 같은 배율을 ffmpeg atempo에 쓴다.
speed=5

# 상세 정도: 1 / 2 / 3
#   1 = 한두 문장, 결과만
#   2 = 작업 규모에 따라 2~10문장, 과정과 결정사항 포함
#   3 = 근거, 트레이드오프, 후속 과제까지 상세히
verbosity=2

# 응답 완료 전 발화: on / off
# on이면 질문 선택지 안내와 긴 작업의 중간 phase 보고도 읽는다.
# off면 응답이 끝났을 때의 요약만 읽는다. 상세 정도(verbosity)와는 무관하다.
interim=on

# 음성 프로바이더: say / gemini-api / elevenlabs-api
#   say            macOS 내장 음성. 무료, 오프라인, 추가 설치 없음
#   gemini-api     GEMINI_API_KEY 필요. 유료. 속도 보정 시 ffmpeg 필요
#   elevenlabs-api ELEVENLABS_API_KEY 필요. 유료
# API 프로바이더가 실패하면 say로 자동 폴백한다.
provider=say

# 프로바이더별 음성. 비워 두면 각 프로바이더의 기본값을 쓴다.
# say 음성 이름 예: Yuna (Premium). 설치된 음성 목록은 `say -v '?'`로 확인
voice_say=

# Gemini 음성 예: Puck, Kore, Charon, Aoede, Zephyr
voice_gemini=

# ElevenLabs 음성 이름 또는 voice ID 예: Yuna
voice_elevenlabs=

# Gemini 음성의 언어 코드 예: ko-KR, en-US. 요약 언어와 짝을 맞춘다.
language_code=ko-KR
